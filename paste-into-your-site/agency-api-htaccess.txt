# Copy from HERE down.

# Never show a list of what is in this folder. A listing tells a stranger
# exactly which files to try next.
Options -Indexes

# Refuse the file types that should never be served out of here: backups of
# your pages, and any stray text file such as an old token.txt.
#
# Note what this does NOT do: it never denies .php. The endpoint keeps working
# no matter what, which is the whole point of writing it this way round. An
# .htaccess that denies everything and then re-allows one file is tidier on
# paper and takes the whole folder down with a 500 on some shared hosts.
<FilesMatch "\.(bak|txt|log|sql|zip|tar|gz|json|md|ini|conf|old|orig|save)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>

# Keep this folder out of Google.
<IfModule mod_headers.c>
    Header set X-Robots-Tag "noindex, nofollow, noarchive"
</IfModule>
